import { StepCard, StepShot } from '@components/awflow';

<StepCard n={1} title="Click Forgot your password?">
  It sits under the **Password** field of the login screen.

  <StepShot shot="login" spot="forgot" n={1} />
</StepCard>

<StepCard n={2} title="Enter your email and click Submit" check="you see Password reset link sent.">
  Use the email address of your account.

  <StepShot shot="forgot-password" spot="submit" n={2} zoom={40} />
</StepCard>

<StepCard n={3} title="Open the email and click the reset link">
  The link opens the reset page in AWFlow.
</StepCard>

<StepCard n={4} title="Choose a new password" check="you see Password has been reset.">
  Enter a **New Password** and **Confirm Password**, then click **Reset Password**.

  <StepShot shot="reset-password" spot="reset" n={4} zoom={40} />
</StepCard>

<StepCard n={5} title="Log in">
  Click **Login to your account** and log in with the new password.
</StepCard>

If the reset page says *Unexpected Error*, the link is incomplete or out of date. Ask for a new one.

To change a password you still know, use [Settings › General](/app/settings/general/#password) instead.
